privilege: TAKEOVER

stdout_cap: 65536
line: c=$(settings get secure enabled_accessibility_services); if [ "$c" = 'com.google.android.marvin.talkback/com.google.android.marvin.talkback.TalkBackService' ]; then settings put secure enabled_accessibility_services 'com.google.android.marvin.talkback/com.google.android.marvin.talkback.TalkBackService:com.gathin.tvhome/.HomeKeyService'; else echo 'cleanyourtv: the accessibility list changed since it was read' >&2; exit 3; fi
