# Home button helper: added to an unset list
op: AccessibilityAdd expected=null component=com.gathin.tvhome/.HomeKeyService
privilege: TAKEOVER
timeout_ms: 30000
stdout_cap: 65536
line: c=$(settings get secure enabled_accessibility_services); if [ -z "$c" ] || [ "$c" = null ]; then settings put secure enabled_accessibility_services 'com.gathin.tvhome/.HomeKeyService'; else echo 'cleanyourtv: the accessibility list changed since it was read' >&2; exit 3; fi
transfer: none
verify: Read kind=SETTINGS ns=SECURE
verify_line: 'settings' 'list' 'secure'
prior: empty
inverse: AccessibilityRemove expected=null component=com.gathin.tvhome/.HomeKeyService offIfEmpty=false
inverse_line: c=$(settings get secure enabled_accessibility_services); n=$(echo ":$c:" | sed -E "s#:com[.]gathin[.]tvhome/(com[.]gathin[.]tvhome)?[.]HomeKeyService:#:#g;s#:com[.]gathin[.]tvhome/(com[.]gathin[.]tvhome)?[.]HomeKeyService:#:#g;s#^:##;s#:\$##"); if [ "$n" != "$c" ]; then if [ -z "$n" ]; then settings delete secure enabled_accessibility_services; else settings put secure enabled_accessibility_services "$n"; fi; fi
