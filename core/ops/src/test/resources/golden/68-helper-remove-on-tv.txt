# Home button helper: taken out of the list the TV has now, accessibility off when it was the last
op: AccessibilityRemove expected=null component=com.gathin.tvhome/.HomeKeyService offIfEmpty=true
privilege: PROFILE_STEP
timeout_ms: 30000
stdout_cap: 65536
line: c=$(settings get secure enabled_accessibility_services); n=$(echo ":$c:" | sed -E "s#:com[.]gathin[.]tvhome/(com[.]gathin[.]tvhome)?[.]HomeKeyService:#:#g;s#:com[.]gathin[.]tvhome/(com[.]gathin[.]tvhome)?[.]HomeKeyService:#:#g;s#^:##;s#:\$##"); if [ "$n" != "$c" ]; then if [ -z "$n" ]; then settings delete secure enabled_accessibility_services; settings put secure accessibility_enabled 0; else settings put secure enabled_accessibility_services "$n"; fi; fi
transfer: none
verify: Read kind=SETTINGS ns=SECURE
verify_line: 'settings' 'list' 'secure'
prior: none
inverse: none
